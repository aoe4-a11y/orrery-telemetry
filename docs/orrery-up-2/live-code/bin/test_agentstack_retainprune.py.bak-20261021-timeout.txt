def put(e, change):
    value = copy.deepcopy(e['value'])
    change(value)
    e['manifest'].write_text(json.dumps(value))
    return e['manifest'].read_bytes()


def test_healthy_no_write_no_mail(env):
    before = env['manifest'].stat().st_mtime_ns
    assert env['run']()['kind'] == 'healthy'
    assert env['manifest'].stat().st_mtime_ns == before
    assert not env['notices']
    assert not list(env['root'].glob('*.bak-*'))


@pytest.mark.parametrize('field', p.FIELDS)
def test_missing_prunes_only_absent_with_real_governor(env, field):
    absent = str(env['root'] / 'absent')
    before = put(env, lambda value: value[field].append(absent))
    result = env['run']()
    assert result['kind'] == 'pruned'
    assert result['count'] == 1
    assert env['keep'].read_bytes() == b'protected unchanged'
    value = json.loads(env['manifest'].read_bytes())
    assert absent not in value[field]
    assert value['protected_paths'] == [str(env['keep'])]
    assert value['scope']['existing'] == 'preserve'
    assert Path(result['backup']).read_bytes() == before
    assert p.sha(env['manifest'].read_bytes()) == result['after_sha256']
    assert json.loads(Path(result['audit']).read_bytes())['phase'] == 'applied'
    p.validator(json.loads(env['policy'].read_bytes()))
    assert env['run']()['kind'] == 'healthy'
    assert len(env['notices']) == 1

def test_permission_denial_not_missing(env):
    before = put(env, lambda value: value['protected_paths'].append(str(env['root'] / 'absent')))
    real = Path.lstat
    def denied(path, *args, **kwargs):
        if path.name == 'absent':
            raise PermissionError(errno.EACCES, 'denied', str(path))
        return real(path, *args, **kwargs)
    with patch.object(Path, 'lstat', denied):
        assert env['run']()['error_type'] == 'PermissionError'
    assert env['manifest'].read_bytes() == before
